下面按“日常最常用”的顺序整理一份 MySQL 速查表。示例中假设数据库为 `school`，表为 `students`。

# 登录
```
-- 登录
mysql -u root -p

-- 查看数据库 / 选择数据库
SHOW DATABASES;
CREATE DATABASE school CHARACTER SET utf8mb4;
USE school;
DROP DATABASE school;  -- 删除数据库，谨慎
```
# 建表

```
-- 建表
CREATE TABLE students (
  id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(50) NOT NULL,
  age TINYINT UNSIGNED,
  score DECIMAL(5,2) DEFAULT 0,
  email VARCHAR(100) UNIQUE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

SHOW TABLES;
DESC students;
SHOW CREATE TABLE students;
```
# 插入

```
-- 插入
INSERT INTO students (name, age, score, email)
VALUES ('张三', 22, 88.5, 'zhangsan@example.com');

-- 一次插入多行
INSERT INTO students (name, age, score)
VALUES
  ('李四', 21, 91.0),
  ('王五', 23, 76.5);
```

# 查询

```
-- 查询
SELECT * FROM students;

-- 指定列、别名
SELECT name AS 姓名, score AS 成绩
FROM students;

-- 条件
SELECT * FROM students
WHERE age >= 22 AND score >= 80;

-- 模糊查询：% 任意长度，_ 一个字符
SELECT * FROM students WHERE name LIKE '张%';

-- 范围 / 集合 / 空值
SELECT * FROM students WHERE score BETWEEN 60 AND 90;
SELECT * FROM students WHERE age IN (20, 21, 22);
SELECT * FROM students WHERE email IS NULL;

-- 排序、分页
SELECT * FROM students
ORDER BY score DESC, age ASC
LIMIT 10 OFFSET 0;  -- 前 10 条

-- 去重
SELECT DISTINCT age FROM students;
```

# 更新
```
-- 更新
UPDATE students
SET score = 95.0, age = 23
WHERE id = 1;

-- 删除：务必先确认 WHERE
DELETE FROM students WHERE id = 1;

-- 清空表（更快，通常无法按行回滚）
TRUNCATE TABLE students;
```


# 修改表结构
```
-- 修改表结构
ALTER TABLE students ADD COLUMN phone VARCHAR(20);
ALTER TABLE students MODIFY COLUMN name VARCHAR(100) NOT NULL;
ALTER TABLE students CHANGE COLUMN phone mobile VARCHAR(20);
ALTER TABLE students DROP COLUMN mobile;

ALTER TABLE students ADD INDEX idx_name (name);
ALTER TABLE students DROP INDEX idx_name;
DROP TABLE students;  -- 删除表，谨慎
```
# 聚合统计

```
-- 聚合统计
SELECT COUNT(*) FROM students;
SELECT AVG(score), MAX(score), MIN(score), SUM(score) FROM students;

-- 分组：HAVING 用于筛选分组结果
SELECT age, COUNT(*) AS total, AVG(score) AS avg_score
FROM students
GROUP BY age
HAVING avg_score >= 80;
```
